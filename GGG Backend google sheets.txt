/**

 * addSubmission() maps to headers dynamically so column order doesn't matter.

const EXPECTED_COLUMNS = [
  'id', 'name', 'email', 'rideDate', 'elapsedTime', 'avgSpeed', 'totalDistance', 'movingTime',
  'stravaLink', 'bonuses', 'foodOrders', 'ebikeUsed', 'comments', 'costcosVisited',

/**
 * Verify admin token from Script Properties
 */
function verifyAdminToken(token) {
  if (!token) return false;
  const scriptProperties = PropertiesService.getScriptProperties();
  const storedToken = scriptProperties.getProperty('ADMIN_TOKEN');
  return storedToken && token === storedToken;
}

/**
 * Handle GET requests
 */
function doGet(e) {
  const action = e.parameter.action;

  try {
    let result;

    switch (action) {
      case 'getResults':
        // Public endpoint - no auth required
        const year = e.parameter.year || new Date().getFullYear();
        result = getApprovedResults(parseInt(year));
        break;

      case 'getSubmissions':
        // Admin endpoint - requires valid admin token
        const adminToken = e.parameter.adminToken;
        if (verifyAdminToken(adminToken)) {
          result = getAllSubmissions();
        } else {
          result = { error: 'Unauthorized' };
        }
        break;

      default:
        result = { error: 'Unknown action' };
    }

    return ContentService.createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ error: error.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * Handle POST requests
 */
function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    let result;

    if (data.action === 'updateStatus') {
      // Admin action - requires valid admin token
      if (verifyAdminToken(data.adminToken)) {
        result = updateSubmissionStatus(data.id, data.status);
      } else {
        result = { error: 'Unauthorized' };
      }
    } else if (data.action === 'uploadGpx') {
      // GPX file upload - separate from form submission
      result = uploadGpxFile(data);
    } else {
      // Public submission - no auth required
      result = addSubmission(data);
    }

    return ContentService.createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ error: error.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * Get the submissions sheet (standalone — uses openById)
 * Creates the sheet with all expected columns if it doesn't exist yet.
 */
function getSheet() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet = ss.getSheetByName(SHEET_NAME);

  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.getRange(1, 1, 1, EXPECTED_COLUMNS.length).setValues([EXPECTED_COLUMNS]);
    sheet.getRange(1, 1, 1, EXPECTED_COLUMNS.length).setFontWeight('bold').setBackground('#f0f0f0');
    sheet.setFrozenRows(1);
  }

  return sheet;
}

/**
 * Add a new submission.
 * Uses header-based column lookup so new columns never break existing data.
 */
function addSubmission(data) {
  const sheet = getSheet();
  const id = Date.now();

  // Check for duplicate GPX hash
  const existingHashes = getExistingHashes();
  const isDuplicate = existingHashes.includes(data.gpxHash);

  // Get current headers from sheet
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];

  // Build a map of all field values
  const fieldValues = {
    id:               id,
    name:             data.name || '',
    email:            data.email || '',
    rideDate:         data.rideDate || '',
    elapsedTime:      data.elapsedTime || '',
    avgSpeed:         data.avgSpeed || '',
    totalDistance:    data.totalDistance || '',
    movingTime:       data.movingTime || '',
    stravaLink:       data.stravaLink || '',
    bonuses:          JSON.stringify(data.bonuses || []),
    foodOrders:       data.foodOrders || '',
    ebikeUsed:        data.ebikeUsed || false,
    comments:         data.comments || '',
    costcosVisited:   data.costcosVisited || 0,
    gpxHash:          data.gpxHash || '',
    gpxDriveUrl:      '',   // filled by separate uploadGpx call
    gpxDriveId:       '',   // filled by separate uploadGpx call
    consentGiven:     data.consentGiven || false,
    verificationData: data.verificationData || '',
    submittedAt:      data.submittedAt || new Date().toISOString(),
    status:           'pending',
    year:             data.year || new Date().getFullYear(),
    reviewedAt:       ''
  };

  // Build row array aligned to current sheet headers
  const row = headers.map(header => {
    return fieldValues.hasOwnProperty(header) ? fieldValues[header] : '';
  });

  sheet.appendRow(row);
  sendNotificationEmail(data, isDuplicate);
  sendConfirmationEmail(data);          // rider confirmation

  return {
    success: true,
    id: id,
    message: 'Submission received',
    possibleDuplicate: isDuplicate
  };
}

/**
 * Upload GPX file to Google Drive and update the matching submission row
 * Called separately from addSubmission to avoid payload size issues
 */
function uploadGpxFile(data) {
  if (!data.gpxFileBase64) {
    return { success: false, message: 'No GPX file data received' };
  }

  try {
    console.log('GPX upload: base64 length = ' + data.gpxFileBase64.length);

    // Decode and create file
    const gpxBytes = Utilities.base64Decode(data.gpxFileBase64);
    const fileName = (data.riderName || 'Unknown').replace(/[^a-zA-Z0-9]/g, '_')
      + '_' + (data.rideDate || 'nodate')
      + '_' + Date.now() + '.gpx';
    const blob = Utilities.newBlob(gpxBytes, 'application/gpx+xml', fileName);

    // Save to Drive
    const folderId = getOrCreateGpxFolder();
    const folder = DriveApp.getFolderById(folderId);
    const file = folder.createFile(blob);
    const gpxDriveUrl = file.getUrl();
    const gpxDriveId = file.getId();

    console.log('GPX file saved: ' + gpxDriveUrl);

    // Find the matching submission row by gpxHash and update it
    if (data.gpxHash) {
      const sheet = getSheet();
      const sheetData = sheet.getDataRange().getValues();
      const headers = sheetData[0];
      const hashCol = headers.indexOf('gpxHash');
      const urlCol = headers.indexOf('gpxDriveUrl');
      const idCol = headers.indexOf('gpxDriveId');

      if (hashCol !== -1 && urlCol !== -1 && idCol !== -1) {
        for (let i = sheetData.length - 1; i >= 1; i--) {
          if (sheetData[i][hashCol] === data.gpxHash) {
            sheet.getRange(i + 1, urlCol + 1).setValue(gpxDriveUrl);
            sheet.getRange(i + 1, idCol + 1).setValue(gpxDriveId);
            console.log('Updated row ' + (i + 1) + ' with Drive link');
            break;
          }
        }
      }
    }

    return { success: true, message: 'GPX uploaded', driveUrl: gpxDriveUrl };

  } catch (e) {
    console.log('GPX upload error: ' + e.toString());
    console.log('Stack: ' + e.stack);
    return { success: false, message: 'GPX upload failed: ' + e.toString() };
  }
}

/**
 * Get existing GPX hashes for duplicate detection
 */
function getExistingHashes() {
  const sheet = getSheet();
  const data = sheet.getDataRange().getValues();

  if (data.length <= 1) return [];

  const hashColumn = data[0].indexOf('gpxHash');
  const statusColumn = data[0].indexOf('status');

  return data.slice(1)
    .filter(row => row[statusColumn] === 'approved')
    .map(row => row[hashColumn])
    .filter(hash => hash);
}

/**
 * Get all submissions (admin)
 */
function getAllSubmissions() {
  const sheet = getSheet();
  const data = sheet.getDataRange().getValues();

  if (data.length <= 1) return [];

  const headers = data[0];

  return data.slice(1).map(row => {
    const obj = {};
    headers.forEach((header, index) => {
      let value = row[index];

      if ((header === 'bonuses') && value) {
        try { value = JSON.parse(value); } catch (e) { value = []; }
      }

      obj[header] = value;
    });
    return obj;
  }).reverse();
}

/**
 * Get approved results (public)
 */
function getApprovedResults(year) {
  const sheet = getSheet();
  const data = sheet.getDataRange().getValues();

  if (data.length <= 1) return [];

  const headers = data[0];
  const statusIndex = headers.indexOf('status');
  const yearIndex = headers.indexOf('year');

  return data.slice(1)
    .filter(row => row[statusIndex] === 'approved' && row[yearIndex] == year)
    .map(row => {
      const obj = {};
      const publicFields = [
        'name', 'rideDate', 'elapsedTime', 'avgSpeed', 'totalDistance', 'movingTime',
        'stravaLink', 'bonuses', 'foodOrders', 'costcosVisited', 'ebikeUsed'
      ];

      headers.forEach((header, index) => {
        if (publicFields.includes(header)) {
          let value = row[index];
          if (header === 'bonuses' && value) {
            try { value = JSON.parse(value); } catch (e) { value = []; }
          }
          obj[header] = value;
        }
      });

      return obj;
    });
}

/**
 * Update submission status
 */
function updateSubmissionStatus(id, newStatus) {
  const sheet = getSheet();
  const data = sheet.getDataRange().getValues();

  const headers = data[0];
  const idIndex = headers.indexOf('id');
  const statusIndex = headers.indexOf('status');
  const reviewedAtIndex = headers.indexOf('reviewedAt');

  for (let i = 1; i < data.length; i++) {
    if (data[i][idIndex] == id) {
      sheet.getRange(i + 1, statusIndex + 1).setValue(newStatus);
      sheet.getRange(i + 1, reviewedAtIndex + 1).setValue(new Date().toISOString());
      sendStatusUpdateEmail(data[i], headers, newStatus);  // notify rider
      return { success: true, message: `Status updated to ${newStatus}` };
    }
  }

  return { success: false, message: 'Submission not found' };
}

/**
 * Send notification email
 */
function sendNotificationEmail(data, isDuplicate) {
  const ADMIN_EMAIL = 'leopoldmarx@gmail.com';

  const subject = `🌭 New GGG Submission: ${data.name}${isDuplicate ? ' ⚠️ POSSIBLE DUPLICATE' : ''}`;

  // Parse food orders for readable display
  let foodOrdersDisplay = 'Not provided';
  if (data.foodOrders) {
    try {
      const orders = JSON.parse(data.foodOrders);
      foodOrdersDisplay = Object.entries(orders)
        .map(([stop, item]) => `  ${stop}: ${item || 'not filled'}`)
        .join('\n');
    } catch (e) {
      foodOrdersDisplay = data.foodOrders;
    }
  }

  const body = `
New Great Glizzy Gauntlet submission received!

Rider: ${data.name}
Email: ${data.email}
Ride Date: ${data.rideDate}
Costcos Visited: ${data.costcosVisited}/10
E-bike: ${data.ebikeUsed ? 'Yes' : 'No'}

Challenge Time: ${data.elapsedTime || 'Not calculated'}
Average Speed: ${data.avgSpeed || 'Not calculated'}
Total Distance: ${data.totalDistance ? data.totalDistance + ' mi' : 'Not calculated'}

Strava Link: ${data.stravaLink || 'Not provided'}
Claimed Bonuses: ${(data.bonuses || []).join(', ') || 'None'}
Comments: ${data.comments || 'None'}
Data Consent: ${data.consentGiven ? 'Yes' : 'No'}

Food Orders:
${foodOrdersDisplay}

${isDuplicate ? '⚠️ WARNING: This GPX hash matches an existing approved submission!' : ''}

Review this submission at: https://ggg.lmarx.com/admin
  `;

  try {
    MailApp.sendEmail(ADMIN_EMAIL, subject, body);
  } catch (e) {
    console.log('Could not send notification email:', e);
  }
}

/**
 * =====================================================
 * generateColumns() — RUN THIS after any code update
 * =====================================================
 * Checks the sheet for all expected columns and adds
 * any that are missing. Safe to run on a sheet with
 * existing data — never deletes or reorders columns.
 *
 * Returns a summary of what was added (or confirms
 * everything is already in place).
 */
function generateColumns() {
  const sheet = getSheet();
  const currentHeaders = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];

  const missing = EXPECTED_COLUMNS.filter(col => !currentHeaders.includes(col));

  if (missing.length === 0) {
    console.log('✅ All columns present: ' + EXPECTED_COLUMNS.join(', '));
    return '✅ All ' + EXPECTED_COLUMNS.length + ' columns already present — nothing to add.';
  }

  // Append missing columns at the end
  const startCol = currentHeaders.length + 1;
  missing.forEach((colName, i) => {
    const col = startCol + i;
    sheet.getRange(1, col).setValue(colName).setFontWeight('bold').setBackground('#f0f0f0');
    sheet.setColumnWidth(col, 180);
  });

  const msg = `✅ Added ${missing.length} missing column(s): ${missing.join(', ')}`;
  console.log(msg);
  return msg;
}

/**
 * Get the Google Drive folder ID for storing GPX files
 * Folder must be created manually and ID set in Script Properties (GPX_FOLDER_ID)
 */
function getOrCreateGpxFolder() {
  const scriptProperties = PropertiesService.getScriptProperties();
  const folderId = scriptProperties.getProperty('GPX_FOLDER_ID');
  return folderId;
}

/**
 * STEP 1: Run this from the editor to authorize Drive + Sheets access.
 * It will:
 *   - Prompt you to authorize Drive and Sheets permissions
 *   - Verify it can access the GPX folder (set GPX_FOLDER_ID in Script Properties first)
 *   - Verify it can access the spreadsheet
 *
 * After running successfully, deploy a new version of the web app.
 */
function authorizeDriveAccess() {
  // Test Drive access
  const rootFolder = DriveApp.getRootFolder();
  console.log('Drive access authorized! Root folder: ' + rootFolder.getName());

  // Test GPX folder access
  const scriptProperties = PropertiesService.getScriptProperties();
  const folderId = scriptProperties.getProperty('GPX_FOLDER_ID');
  if (folderId) {
    const folder = DriveApp.getFolderById(folderId);
    console.log('GPX folder accessible: ' + folder.getName() + ' (ID: ' + folderId + ')');
  } else {
    console.log('WARNING: GPX_FOLDER_ID not set in Script Properties! Set it before deploying.');
  }

  // Test Spreadsheet access
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  console.log('Spreadsheet accessible: ' + ss.getName());

  const sheet = ss.getSheetByName(SHEET_NAME);
  if (sheet) {
    console.log('Sheet "' + SHEET_NAME + '" found with ' + sheet.getLastRow() + ' rows');
  } else {
    console.log('Sheet "' + SHEET_NAME + '" not found — run generateColumns() or check SHEET_NAME');
  }

  return 'All access verified! Ready to deploy.';
}
